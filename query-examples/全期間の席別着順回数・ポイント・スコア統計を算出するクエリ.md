## 説明
全期間の席別着順回数・ポイント・スコア統計を算出するクエリ

## SQL クエリ
```sql
-- 各ゲームでの席（東1局本場なしの自風）を取得
WITH game_seat AS (
    SELECT
        k.game_id,
        kpr.player_id,
        kpr.player_wind AS seat
    FROM kyoku_player_result kpr
    -- 東1局・本場なしの局を特定
    JOIN kyoku k ON kpr.kyoku_id = k.id
    WHERE k.round = '1z1' AND k.honba_count = 0
),
-- 席別の通算成績を集計
seat_stats AS (
    SELECT
        gs.seat,
        COUNT(*) AS total_games,
        -- 着順回数の集計
        SUM(CASE WHEN gpr.rank = 1 THEN 1 ELSE 0 END) AS rank1_count,
        SUM(CASE WHEN gpr.rank = 2 THEN 1 ELSE 0 END) AS rank2_count,
        SUM(CASE WHEN gpr.rank = 3 THEN 1 ELSE 0 END) AS rank3_count,
        SUM(CASE WHEN gpr.rank = 4 THEN 1 ELSE 0 END) AS rank4_count,
        -- ポイント・スコアの合計
        SUM(gpr.points) AS total_points,
        SUM(gpr.score) AS total_score
    FROM game_seat gs
    -- 試合結果の結合
    JOIN game_player_result gpr ON gs.game_id = gpr.game_id AND gs.player_id = gpr.player_id
    GROUP BY gs.seat
)
-- 最終結果: 席別通算成績を出力
SELECT
    CASE seat
        WHEN '1z' THEN '東席'
        WHEN '2z' THEN '南席'
        WHEN '3z' THEN '西席'
        WHEN '4z' THEN '北席'
    END AS 席,
    total_games AS 総ゲーム数,
    rank1_count AS 一着回数,
    rank2_count AS 二着回数,
    rank3_count AS 三着回数,
    rank4_count AS 四着回数,
    ROUND(rank1_count * 100.0 / total_games, 2) AS トップ率,
    total_points AS トータルポイント,
    total_score AS トータルスコア
FROM seat_stats
ORDER BY seat;
```
