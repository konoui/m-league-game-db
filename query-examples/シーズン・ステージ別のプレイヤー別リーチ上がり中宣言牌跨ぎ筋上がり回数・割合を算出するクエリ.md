## 説明
シーズン・ステージ別のプレイヤー別リーチ上がり中宣言牌跨ぎ筋上がり回数・割合を算出するクエリ

## SQL クエリ
```sql
-- 受理されたリーチ宣言打牌とその宣言牌を取得する
WITH reach_with_decl AS (
    SELECT
        re.actor_player_id,
        e_re.kyoku_id,
        -- 宣言牌の数値を取得（赤ドラ 0 → 5 に正規化）
        CASE SUBSTR(de.tile, 1, 1) WHEN '0' THEN 5 ELSE CAST(SUBSTR(de.tile, 1, 1) AS INTEGER) END AS decl_num,
        SUBSTR(de.tile, 2, 1) AS decl_suit
    FROM reach_event re
    -- 局 ID の取得
    JOIN event e_re ON re.event_id = e_re.id
    -- 同局・同プレイヤーのリーチ宣言打牌を結合
    JOIN discard_event de ON de.actor_player_id = re.actor_player_id AND de.is_reach_declaration = 1
    JOIN event e_de ON de.event_id = e_de.id AND e_de.kyoku_id = e_re.kyoku_id
    WHERE re.is_accepted = 1
),
-- 同局でリーチプレイヤーが上がった場合の上がり牌を取得する
reach_agari AS (
    SELECT
        rwd.*,
        -- 上がり牌の赤ドラ正規化
        CASE ae.winning_tile WHEN '0m' THEN '5m' WHEN '0p' THEN '5p' WHEN '0s' THEN '5s' ELSE ae.winning_tile END AS norm_win
    FROM reach_with_decl rwd
    -- 同局で同プレイヤーがあがったイベントを結合
    JOIN agari_event ae ON ae.actor_player_id = rwd.actor_player_id
    JOIN event e_ae ON ae.event_id = e_ae.id AND e_ae.kyoku_id = rwd.kyoku_id
),
-- 上がり牌が宣言牌の跨ぎ筋か判定する
matagi_check AS (
    SELECT *,
        CASE
            -- 字牌は跨ぎ筋なし
            WHEN decl_suit NOT IN ('m', 'p', 's') THEN 0
            -- [(X-1) と (X-1)+3]: X-1 と X+2（X が 2〜7 のとき有効）
            WHEN decl_num BETWEEN 2 AND 7 AND norm_win = CAST(decl_num - 1 AS TEXT) || decl_suit THEN 1
            WHEN decl_num BETWEEN 2 AND 7 AND norm_win = CAST(decl_num + 2 AS TEXT) || decl_suit THEN 1
            -- [(X+1) と (X+1)-3]: X+1 と X-2（X が 3〜8 のとき有効）
            WHEN decl_num BETWEEN 3 AND 8 AND norm_win = CAST(decl_num + 1 AS TEXT) || decl_suit THEN 1
            WHEN decl_num BETWEEN 3 AND 8 AND norm_win = CAST(decl_num - 2 AS TEXT) || decl_suit THEN 1
            ELSE 0
        END AS is_matagi
    FROM reach_agari
)
-- シーズン・ステージ・プレイヤー別に集計する
SELECT
    ls.start_year || '-' || ls.end_year AS シーズン,
    ss.stage AS ステージ,
    p.name AS プレイヤー名,
    t.name AS チーム名,
    COUNT(*) AS リーチ上がり数,
    SUM(is_matagi) AS 跨ぎ筋上がり数,
    ROUND(SUM(is_matagi) * 100.0 / COUNT(*), 2) AS 跨ぎ筋上がり割合
FROM matagi_check mc
-- 局・試合・シーズン情報の結合
JOIN kyoku k ON mc.kyoku_id = k.id
JOIN game g ON k.game_id = g.id
JOIN season_stage ss ON g.season_stage_id = ss.id
JOIN league_season ls ON ss.league_season_id = ls.id
-- プレイヤー情報の結合
JOIN player p ON mc.actor_player_id = p.id
-- チーム所属情報の結合
JOIN player_team pt ON mc.actor_player_id = pt.player_id
    AND ls.start_year >= pt.joined_season_year
    AND ls.start_year <= pt.left_season_year
JOIN team t ON pt.team_id = t.id
GROUP BY ls.id, ss.id, mc.actor_player_id
ORDER BY ls.start_year DESC, ss.stage, 跨ぎ筋上がり割合 DESC;
```
