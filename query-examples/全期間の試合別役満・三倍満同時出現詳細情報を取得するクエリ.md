## 説明
全期間の試合別役満・三倍満同時出現詳細情報を取得するクエリ

## SQL クエリ
```sql
-- 同一試合内の役満を集計
WITH yakuman_in_game AS (
    SELECT
        k.game_id,
        GROUP_CONCAT(p.name || '(' || ae.description || '・' || ae.points || '点)', '、') AS yakuman_detail
    FROM agari_event ae
    JOIN event e ON ae.event_id = e.id
    JOIN kyoku k ON e.kyoku_id = k.id
    -- あがったプレイヤー情報の結合
    JOIN player p ON ae.actor_player_id = p.id
    WHERE ae.is_yakuman = 1
    GROUP BY k.game_id
),
-- 同一試合内の三倍満（役満以外の11翻以上）を集計
sanbaiman_in_game AS (
    SELECT
        k.game_id,
        GROUP_CONCAT(p.name || '(' || ae.han || '翻・' || ae.points || '点)', '、') AS sanbaiman_detail
    FROM agari_event ae
    JOIN event e ON ae.event_id = e.id
    JOIN kyoku k ON e.kyoku_id = k.id
    -- あがったプレイヤー情報の結合
    JOIN player p ON ae.actor_player_id = p.id
    WHERE ae.is_yakuman = 0 AND ae.han >= 11
    GROUP BY k.game_id
)
-- 役満と三倍満が同一試合に出現した試合を取得
SELECT
    ls.start_year || '-' || (ls.start_year + 1) AS シーズン,
    ss.stage AS ステージ,
    g.date AS 試合日,
    g.round_number AS ラウンド番号,
    y.yakuman_detail AS 役満,
    s.sanbaiman_detail AS 三倍満
FROM yakuman_in_game y
-- 三倍満が出た試合との絞り込み結合
JOIN sanbaiman_in_game s ON y.game_id = s.game_id
-- 試合・シーズン情報の結合
JOIN game g ON y.game_id = g.id
JOIN season_stage ss ON g.season_stage_id = ss.id
JOIN league_season ls ON ss.league_season_id = ls.id
ORDER BY g.date DESC;
```

## 参照テーブル

- event
- kyoku
- player
- game
- season_stage
- league_season
