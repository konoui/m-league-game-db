## 説明
全期間のプレイヤー別役満あがり詳細情報を取得するクエリ

## SQL クエリ
```sql
-- 役満のあがりごとに、成立した役満の役名をまとめる
WITH yakuman_yaku AS (
    SELECT
        ay.agari_event_id,
        GROUP_CONCAT(yn.name, '・') AS yaku_names
    FROM agari_yaku ay
    JOIN yaku_name yn ON ay.yaku_name_id = yn.id
    -- 役満の役は13翻以上で記録されている
    WHERE ay.han >= 13
    GROUP BY ay.agari_event_id
)
-- 最終結果: 役満のあがりを新しい順に出力
SELECT
    ls.start_year || '-' || (ls.start_year + 1) AS シーズン,
    ss.stage AS ステージ,
    g.date AS 試合日,
    g.stage_game_number AS ラウンド番号,
    -- 場風と局数を「東1局」の形にする
    CASE SUBSTR(k.round, 1, 2)
        WHEN '1z' THEN '東'
        WHEN '2z' THEN '南'
        WHEN '3z' THEN '西'
        WHEN '4z' THEN '北'
    END || SUBSTR(k.round, 3) || '局' AS 局,
    k.honba_count AS 本場,
    p.name AS プレイヤー名,
    t.name AS チーム名,
    yy.yaku_names AS 役満,
    CASE e.type WHEN 'tsumo' THEN 'ツモ' ELSE 'ロン' END AS あがり方,
    tp.name AS 放銃プレイヤー名,
    ae.agari_points AS 打点
FROM agari_event ae
-- 役満の役名の結合
JOIN yakuman_yaku yy ON ae.event_id = yy.agari_event_id
-- 局・試合・シーズン情報の結合
JOIN event e ON ae.event_id = e.id
JOIN kyoku k ON e.kyoku_id = k.id
JOIN game g ON k.game_id = g.id
JOIN season_stage ss ON g.season_stage_id = ss.id
JOIN league_season ls ON ss.league_season_id = ls.id
-- あがったプレイヤーと、ロンの場合の放銃したプレイヤーの結合
JOIN player p ON ae.actor_player_id = p.id
LEFT JOIN player tp ON ae.target_player_id = tp.id
-- 試合時点の所属チームの結合
JOIN game_player_result gpr ON g.id = gpr.game_id AND p.id = gpr.player_id
JOIN team t ON gpr.team_id = t.id
WHERE ae.is_yakuman = 1
ORDER BY g.date DESC, g.stage_game_number DESC, k.id DESC;
```
