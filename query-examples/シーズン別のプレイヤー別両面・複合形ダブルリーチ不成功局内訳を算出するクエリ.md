## 説明
シーズン別のプレイヤー別両面・複合形ダブルリーチ不成功局内訳を算出するクエリ

## SQL クエリ
```sql
-- ダブル立直かつ両面・複合形待ちのリーチ宣言打牌イベントをユニークに特定する
WITH double_reach_tenpai AS (
    SELECT DISTINCT
        tam.tenpai_event_id,
        de.actor_player_id,
        e.kyoku_id
    FROM tenpai_yaku_event tye
    JOIN tenpai_agari_matrix tam ON tye.tenpai_agari_matrix_id = tam.id
    -- 役名の確認
    JOIN yaku_name yn ON tye.name_id = yn.id
    -- リーチ宣言打牌からプレイヤー ID を取得（リーチ宣言打牌のみに絞る）
    JOIN discard_event de ON de.event_id = tam.tenpai_event_id AND de.is_reach_declaration = 1
    -- 待ちタイプの確認
    JOIN player_tenpai_state pts
        ON pts.event_id = tam.tenpai_event_id AND pts.player_id = de.actor_player_id
    JOIN event e ON e.id = tam.tenpai_event_id
    WHERE yn.name = 'ダブル立直'
      AND pts.waiting_type IN ('両面', '複合形')
),
-- 各局の終了結果を4カテゴリに分類する（自分があがった局は除外）
kyoku_outcome AS (
    SELECT
        drt.actor_player_id,
        drt.kyoku_id,
        CASE
            -- 自分があがった局は集計対象外とする
            WHEN EXISTS (
                SELECT 1 FROM agari_event ae
                JOIN event ev ON ae.event_id = ev.id
                WHERE ev.kyoku_id = drt.kyoku_id
                  AND ae.actor_player_id = drt.actor_player_id
            ) THEN NULL
            -- 流局
            WHEN EXISTS (
                SELECT 1 FROM ryukyoku_event re
                JOIN event ev ON re.event_id = ev.id
                WHERE ev.kyoku_id = drt.kyoku_id
            ) THEN '流局'
            -- ほかの人ツモ: 他プレイヤーのツモあがり
            WHEN EXISTS (
                SELECT 1 FROM agari_event ae
                JOIN event ev ON ae.event_id = ev.id
                WHERE ev.kyoku_id = drt.kyoku_id
                  AND ae.target_player_id IS NULL
                  AND ae.actor_player_id != drt.actor_player_id
            ) THEN 'ほかの人ツモ'
            -- 自分が放銃: 自分の打牌でほかの人がロン
            WHEN EXISTS (
                SELECT 1 FROM agari_event ae
                JOIN event ev ON ae.event_id = ev.id
                WHERE ev.kyoku_id = drt.kyoku_id
                  AND ae.target_player_id = drt.actor_player_id
            ) THEN '自分が放銃'
            -- ほかの人が放銃: 自分が関与しないロン
            WHEN EXISTS (
                SELECT 1 FROM agari_event ae
                JOIN event ev ON ae.event_id = ev.id
                WHERE ev.kyoku_id = drt.kyoku_id
                  AND ae.target_player_id IS NOT NULL
                  AND ae.target_player_id != drt.actor_player_id
                  AND ae.actor_player_id != drt.actor_player_id
            ) THEN 'ほかの人が放銃'
            ELSE NULL
        END AS outcome
    FROM double_reach_tenpai drt
)
-- シーズン別・プレイヤー別に不成功局の内訳を集計する
SELECT
    ls.start_year || '-' || ls.end_year AS シーズン,
    p.name AS プレイヤー名,
    t.name AS チーム名,
    -- 結果カテゴリ別の回数
    SUM(CASE WHEN ko.outcome = '流局' THEN 1 ELSE 0 END) AS 流局,
    SUM(CASE WHEN ko.outcome = 'ほかの人ツモ' THEN 1 ELSE 0 END) AS ほかの人ツモ,
    SUM(CASE WHEN ko.outcome = '自分が放銃' THEN 1 ELSE 0 END) AS 自分が放銃,
    SUM(CASE WHEN ko.outcome = 'ほかの人が放銃' THEN 1 ELSE 0 END) AS ほかの人が放銃,
    COUNT(ko.outcome) AS 不成功合計
FROM kyoku_outcome ko
-- 局・試合・シーズン情報の結合
JOIN kyoku k ON ko.kyoku_id = k.id
JOIN game g ON k.game_id = g.id
JOIN season_stage ss ON g.season_stage_id = ss.id
JOIN league_season ls ON ss.league_season_id = ls.id
-- プレイヤー情報の結合
JOIN player p ON ko.actor_player_id = p.id
-- チーム所属情報の結合
JOIN player_team pt ON ko.actor_player_id = pt.player_id
    AND ls.start_year >= pt.joined_season_year
    AND ls.start_year <= pt.left_season_year
JOIN team t ON pt.team_id = t.id
-- 不成功局のみ集計
WHERE ko.outcome IS NOT NULL
GROUP BY ls.id, ko.actor_player_id
ORDER BY ls.start_year DESC, 不成功合計 DESC;
```
